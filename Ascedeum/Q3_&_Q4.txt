import os
import sys
import time
import uuid
import random
import signal
import subprocess
import threading
import redis

REDIS_URL = os.getenv("REDIS_URL", "redis://localhost:6379/0")
MAX_HEAVY = int(os.getenv("MAX_HEAVY", "3"))
LEASE = int(os.getenv("LEASE", "300"))
KEY = "ultron:heavy_semaphore"

r = redis.Redis.from_url(REDIS_URL, decode_responses=True)
TOKEN = str(uuid.uuid4())

ACQUIRE = """
local now = redis.call('TIME')[1]
redis.call('ZREMRANGEBYSCORE', KEYS[1], '-inf', now)
if redis.call('ZCARD', KEYS[1]) < tonumber(ARGV[1]) then
    redis.call('ZADD', KEYS[1], now + tonumber(ARGV[2]), ARGV[3])
    return 1
end
return 0
"""

RENEW = """
if redis.call('ZSCORE', KEYS[1], ARGV[1]) then
    local now = redis.call('TIME')[1]
    redis.call('ZADD', KEYS[1], now + tonumber(ARGV[2]), ARGV[1])
    return 1
end
return 0
"""

RELEASE = """
return redis.call('ZREM', KEYS[1], ARGV[1])
"""

def acquire():
    while True:
        ok = r.eval(ACQUIRE, 1, KEY, MAX_HEAVY, LEASE, TOKEN)
        if ok:
            return
        time.sleep(1 + random.random() * 2)

def renew(stop):
    while not stop.wait(LEASE / 3):
        if not r.eval(RENEW, 1, KEY, TOKEN, LEASE):
            raise RuntimeError("Lost Redis semaphore lease")

def release():
    r.eval(RELEASE, 1, KEY, TOKEN)

def run(command, heavy):
    if not heavy:
        return subprocess.call(command)

    acquire()
    stop = threading.Event()
    heartbeat = threading.Thread(target=renew, args=(stop,), daemon=True)
    heartbeat.start()

    try:
        return subprocess.call(command)
    finally:
        stop.set()
        release()

def main():
    if len(sys.argv) < 3:
        print("Usage: wrapper.py <heavy|light> <command...>")
        return 2

    heavy = sys.argv[1].lower() == "heavy"

    def stop_handler(signum, frame):
        raise KeyboardInterrupt

    signal.signal(signal.SIGTERM, stop_handler)
    signal.signal(signal.SIGINT, stop_handler)

    try:
        return run(sys.argv[2:], heavy)
    except KeyboardInterrupt:
        return 143

if __name__ == "__main__":
    sys.exit(main())
